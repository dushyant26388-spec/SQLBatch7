1. Find employees who earn more than the average salary
SELECT name, salary
FROM employees
WHERE salary > (
    SELECT AVG(salary)
    FROM employees
);
2. Find the employee with the highest salary
SELECT name, salary
FROM employees
WHERE salary = (
    SELECT MAX(salary)
    FROM employees
);
3. Find the employee with the second highest salary
SELECT name, salary
FROM employees
WHERE salary = (
    SELECT MAX(salary)
    FROM employees
    WHERE salary < (
        SELECT MAX(salary)
        FROM employees
    )
);
4. Find employees who earn less than the average salary
SELECT name, salary
FROM employees
WHERE salary < (
    SELECT AVG(salary)
    FROM employees
);
5. Find employees working in the same department as Amit
SELECT name, department
FROM employees
WHERE department = (
    SELECT department
    FROM employees
    WHERE name = 'Amit'
);
6. Find employees who earn more than Amit
SELECT name, salary
FROM employees
WHERE salary > (
    SELECT salary
    FROM employees
    WHERE name = 'Amit'
);
7. Find employees with the minimum salary
SELECT name, salary
FROM employees
WHERE salary = (
    SELECT MIN(salary)
    FROM employees
);
8. Find all employees working in departments where someone earns more than 55,000
SELECT name, department, salary
FROM employees
WHERE department IN (
    SELECT department
    FROM employees
    WHERE salary > 55000
);
9. Find employees whose salary is equal to any HR employee's salary
SELECT name, salary
FROM employees
WHERE salary IN (
    SELECT salary
    FROM employees
    WHERE department = 'HR'
);
10. Find employees who earn more than every HR employee
SELECT name, salary
FROM employees
WHERE salary > ALL (
    SELECT salary
    FROM employees
    WHERE department = 'HR'
);

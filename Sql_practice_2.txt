6. Find employees whose salary is between 50,000 and 70,000.
SELECT emp_name, salary
FROM employees
WHERE salary BETWEEN 50000 AND 70000;

7. Find the department with the highest average salary.
SELECT department, AVG(salary) AS avg_salary
FROM employees
GROUP BY department
ORDER BY avg_salary DESC
LIMIT 1;

8. Find duplicate salaries.
SELECT salary, COUNT(*) AS salary_count
FROM employees
GROUP BY salary
HAVING COUNT(*) > 1;

9. Find employees whose name starts with 'A'.
SELECT emp_name
FROM employees
WHERE emp_name LIKE 'A%';

10. Find the top 3 highest distinct salaries.
SELECT DISTINCT salary
FROM employees
ORDER BY salary DESC
LIMIT 3;

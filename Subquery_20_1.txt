-- Q1. Find the names of customers who have placed at least one order.

SELECT customer_name
FROM customers
WHERE customer_id IN (
    SELECT customer_id
    FROM orders
);
-- Q2. Find products that have NEVER been ordered.

SELECT product_name
FROM products
WHERE product_id NOT IN (
    SELECT product_id
    FROM orders
);
-- Q3. Find customers whose credit limit is above the average credit limit of all customers.

SELECT customer_name, credit_limit
FROM customers
WHERE credit_limit > (
    SELECT AVG(credit_limit)
    FROM customers
);
-- Q4. Find the most expensive product.

SELECT *
FROM products
WHERE price = (
    SELECT MAX(price)
    FROM products
);
-- Q5. Find orders where the product's price is above 10000.

SELECT *
FROM orders
WHERE product_id IN (
    SELECT product_id
    FROM products
    WHERE price > 10000
);

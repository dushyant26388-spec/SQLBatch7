1. Employees earning more than average salary
SELECT emp_name, salary
FROM employees
WHERE salary > (
    SELECT AVG(salary)
    FROM employees
);
2. Employee with the highest salary
SELECT emp_name, salary
FROM employees
WHERE salary = (
    SELECT MAX(salary)
    FROM employees
);
3. Employee with the lowest salary
SELECT emp_name, salary
FROM employees
WHERE salary = (
    SELECT MIN(salary)
    FROM employees
);
4. Employees working in the IT department
SELECT emp_name, salary
FROM employees
WHERE dept_id = (
    SELECT dept_id
    FROM departments
    WHERE dept_name = 'IT'
);
5. Employees earning more than the HR department's average salary
SELECT emp_name, salary
FROM employees
WHERE salary > (
    SELECT AVG(e.salary)
    FROM employees e
    JOIN departments d
    ON e.dept_id = d.dept_id
    WHERE d.dept_name = 'HR'
);
6. Employees earning the same salary as someone else
SELECT emp_name, salary
FROM employees
WHERE salary IN (
    SELECT salary
    FROM employees
    GROUP BY salary
    HAVING COUNT(*) > 1
);
7. Employees whose salary is greater than the minimum salary
SELECT emp_name, salary
FROM employees
WHERE salary > (
    SELECT MIN(salary)
    FROM employees
);
8. Employees whose salary is less than the maximum salary
SELECT emp_name, salary
FROM employees
WHERE salary < (
    SELECT MAX(salary)
    FROM employees
);
9. Employees working in departments located in Mumbai
SELECT emp_name, dept_id
FROM employees
WHERE dept_id IN (
    SELECT dept_id
    FROM departments
    WHERE location = 'Mumbai'
);
10. Employees earning more than at least one employee

Using ANY:

SELECT emp_name, salary
FROM employees
WHERE salary > ANY (
    SELECT salary
    FROM employees
);

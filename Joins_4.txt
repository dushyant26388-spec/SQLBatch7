16. Show employees working on projects.
Answer
SELECT e.emp_name, d.dept_name, p.project_name
FROM employees e
INNER JOIN departments d
ON e.dept_id = d.dept_id
INNER JOIN projects p
ON d.dept_id = p.dept_id;

17. Show all projects with their department names.
Answer
SELECT p.project_name, d.dept_name
FROM projects p
INNER JOIN departments d
ON p.dept_id = d.dept_id;

18. Show all projects, including projects whose department has no employees.
Answer
SELECT p.project_name, d.dept_name, e.emp_name
FROM projects p
INNER JOIN departments d
ON p.dept_id = d.dept_id
LEFT JOIN employees e
ON d.dept_id = e.dept_id;

19. Show employees and projects of their department, including employees with no project.
Answer
SELECT e.emp_name, p.project_name
FROM employees e
LEFT JOIN projects p
ON e.dept_id = p.dept_id;


20. Find employees who work in departments located in Mumbai.
Answer
SELECT e.emp_name, d.dept_name, d.location
FROM employees e
INNER JOIN departments d
ON e.dept_id = d.dept_id
WHERE d.location = 'Mumbai';

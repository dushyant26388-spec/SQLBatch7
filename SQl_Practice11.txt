Q11. Find employees working in the IT department using a subquery.
SELECT emp_name
FROM employees
WHERE dept_id = (
    SELECT dept_id
    FROM departments
    WHERE dept_name = 'IT'
);

Q12. Find employees whose salary is greater than Aman’s salary.
SELECT emp_name, salary
FROM employees
WHERE salary > (
    SELECT salary
    FROM employees
    WHERE emp_name = 'Aman'
);

Q13. Find the second-highest salary.

Simple subquery method:

SELECT MAX(salary) AS second_highest
FROM employees
WHERE salary < (
    SELECT MAX(salary)
    FROM employees
);

Q14. Find employees who earn more than their department's average salary.

This is a correlated subquery.

SELECT e.emp_name, e.salary, e.dept_id
FROM employees e
WHERE e.salary > (
    SELECT AVG(e2.salary)
    FROM employees e2
    WHERE e2.dept_id = e.dept_id
);

Q15. Find employees who work in departments having more than 1 employee.
SELECT emp_name, dept_id
FROM employees
WHERE dept_id IN (
    SELECT dept_id
    FROM employees
    GROUP BY dept_id
    HAVING COUNT(*) > 1
);

Q16. Find employees whose salary is greater than the average salary of the company, and display their department name.
SELECT e.emp_name,
       e.salary,
       d.dept_name
FROM employees e
JOIN departments d
ON e.dept_id = d.dept_id
WHERE e.salary > (
    SELECT AVG(salary)
    FROM employees
);

Q17. Find the department with the highest average salary.
SELECT d.dept_name, AVG(e.salary) AS avg_salary
FROM employees e
JOIN departments d
ON e.dept_id = d.dept_id
GROUP BY d.dept_name
ORDER BY avg_salary DESC
LIMIT 1;

Q18. Find employees who have the highest salary in their department.
SELECT e.emp_name, e.salary, e.dept_id
FROM employees e
WHERE e.salary = (
    SELECT MAX(e2.salary)
    FROM employees e2
    WHERE e2.dept_id = e.dept_id
);

Q19. Find employees working in departments that exist in the departments table.
SELECT emp_name
FROM employees
WHERE dept_id IN (
    SELECT dept_id
    FROM departments
);

Q20. Find employees whose department exists in the departments table using EXISTS.
SELECT e.emp_name
FROM employees e
WHERE EXISTS (
    SELECT 1
    FROM departments d
    WHERE d.dept_id = e.dept_id
);
